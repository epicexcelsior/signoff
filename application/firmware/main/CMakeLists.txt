set(PR_SECRET_BYTES "0")
set(PR_SECRET_LENGTH 0)
if(DEFINED PROOFREADY_AUTH_SECRET_FILE)
    if(NOT IS_ABSOLUTE "${PROOFREADY_AUTH_SECRET_FILE}")
        message(FATAL_ERROR "Use an absolute private authorization file outside source")
    endif()
    file(REAL_PATH "${PROOFREADY_AUTH_SECRET_FILE}" PR_SECRET_PATH)
    file(REAL_PATH "${CMAKE_SOURCE_DIR}/../.." PR_SOURCE_PATH)
    string(FIND "${PR_SECRET_PATH}" "${PR_SOURCE_PATH}/" PR_SECRET_IN_SOURCE)
    if(PR_SECRET_IN_SOURCE EQUAL 0)
        message(FATAL_ERROR "Keep authorization secrets outside source")
    endif()
    file(READ "${PR_SECRET_PATH}" PR_SECRET_HEX HEX)
    string(LENGTH "${PR_SECRET_HEX}" PR_SECRET_HEX_LENGTH)
    math(EXPR PR_SECRET_LENGTH "${PR_SECRET_HEX_LENGTH} / 2")
    if(PR_SECRET_LENGTH LESS 16 OR PR_SECRET_LENGTH GREATER 128)
        message(FATAL_ERROR "Authorization file must contain 16 through 128 exact secret bytes")
    endif()
    string(REGEX REPLACE "(..)" "0x\\1," PR_SECRET_BYTES "${PR_SECRET_HEX}")
endif()
